cmake_minimum_required(VERSION 3.16)
project(MeowpuutWindows LANGUAGES CXX)
add_subdirectory(../Core core)
add_library(MeowpuutWindowsSession STATIC Session.cpp Keyboard.cpp)
target_link_libraries(MeowpuutWindowsSession PUBLIC MeowpuutCore)
target_compile_features(MeowpuutWindowsSession PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(MeowpuutWindowsSession PRIVATE /utf-8 /W4)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(meowpuut_session_test tests/session_test.cpp)
    target_link_libraries(meowpuut_session_test PRIVATE MeowpuutWindowsSession)
    add_executable(meowpuut_mixed_candidate_test tests/MixedCompositionTests.cpp)
    target_link_libraries(meowpuut_mixed_candidate_test PRIVATE MeowpuutCore)
    add_test(NAME mixed_candidates COMMAND meowpuut_mixed_candidate_test)
    add_executable(meowpuut_pending_delivery_test tests/PendingDeliveryTests.cpp)
    add_test(NAME pending_delivery COMMAND meowpuut_pending_delivery_test)
    add_executable(meowpuut_punctuation_test tests/PinyinPunctuationTests.cpp)
    add_test(NAME pinyin_punctuation COMMAND meowpuut_punctuation_test)
    add_executable(meowpuut_script_mode_test tests/ScriptModeTests.cpp)
    target_link_libraries(meowpuut_script_mode_test PRIVATE MeowpuutCore)
    add_test(NAME script_mode COMMAND meowpuut_script_mode_test)
    if(MSVC)
        target_compile_options(meowpuut_session_test PRIVATE /utf-8 /W4)
        target_compile_options(meowpuut_mixed_candidate_test PRIVATE /utf-8 /W4)
        target_compile_options(meowpuut_pending_delivery_test PRIVATE /utf-8 /W4)
        target_compile_options(meowpuut_punctuation_test PRIVATE /utf-8 /W4)
        target_compile_options(meowpuut_script_mode_test PRIVATE /utf-8 /W4)
    endif()
    set(MEOWPUUT_LEXICON_DIR "" CACHE PATH "Directory containing native-pinyin.tsv, native-zhuyin.tsv and native-japanese.tsv")
    if(MEOWPUUT_LEXICON_DIR)
        add_test(NAME session COMMAND meowpuut_session_test "${MEOWPUUT_LEXICON_DIR}/native-pinyin.tsv" "${MEOWPUUT_LEXICON_DIR}/native-zhuyin.tsv" "${MEOWPUUT_LEXICON_DIR}/native-japanese.tsv")
        add_test(NAME mixed_candidates_real COMMAND meowpuut_mixed_candidate_test "${MEOWPUUT_LEXICON_DIR}/native-pinyin.tsv")
    endif()
endif()
